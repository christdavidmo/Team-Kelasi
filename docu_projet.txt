skula-backend/
├── app/
│   ├── main.py             # Point d'entrée : initialise FastAPI et les routes
│   ├── api/
│   │   └── v1/             # Versioning de l'API (important pour l'évolution)
│   │       ├── api.py      # Routeur principal qui regroupe tous les endpoints
│   │       └── endpoints/  # Un fichier par module (users.py, schools.py, grades.py)
│   ├── core/               # Configuration globale
│   │   ├── config.py       # Gestion des variables d'environnement (.env)
│   │   └── security.py     # Logique JWT et hachage des mots de passe
│   ├── db/                 # Couche d'accès aux données
│   │   ├── base.py         # Import de tous les modèles pour Alembic
│   │   └── session.py      # Configuration de la connexion PostgreSQL
│   ├── models/             # Modèles SQLAlchemy (tes tables en base de données)
│   ├── schemas/            # Schémas Pydantic (validation des entrées/sorties API)
│   └── services/           # LOGIQUE MÉTIER (Calculs de moyennes, validation censeur)
├── alembic/                # Dossier pour les migrations (historique de la DB)
├── tests/                  # Tests automatisés
├── .env                    # Fichier secret (config de ta machine/Docker)
├── .dockerignore           # Fichiers à ne pas envoyer dans l'image Docker
├── Dockerfile              # Recette pour créer ton conteneur Python 3.12
├── docker-compose.yml      # Orchestration (API + PostgreSQL)
└── requirements.txt        # Liste des bibliothèques (FastAPI, SQLAlchemy, etc.)
